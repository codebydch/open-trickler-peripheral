# Manual install reference.
#
# install-part1.sh and install-part2.sh do all of this for you. Keep this file as the
# explanation of what those scripts are doing, or follow it by hand if you would rather.
#
# The scripts differ from these notes in three ways, all deliberate:
#   - Python packages come from requirements-to-freeze.txt rather than being installed
#     one at a time, so there is one list to keep correct.
#   - websocketd is chosen to match the CPU (see below).
#   - the nginx sites-enabled symlink is ensured rather than assumed.
#   - swap is not touched at all; it is left to you, per the README.

# Swap, on a Pi with little RAM (a Pi Zero, Pi Zero 2, older Model A).
# The upgrade below can crash such a board outright. Set swap up first -- see the "Swap"
# section of README.md. Bookworm and earlier use dphys-swapfile; Trixie and later use
# rpi-swap (/etc/rpi/swap.conf, `man swap.conf`), where the defaults do not mean what
# they look like. The install scripts deliberately leave swap alone.
sudo apt update && sudo apt upgrade

# Create code folder at root
cd /
sudo mkdir code
sudo chown pi:pi code
cd code

# Create Python virtual environemnt
cd /code
python3 -m venv venv --system-site-packages
source venv/bin/activate

# Open Trickler install
cd /code
git clone https://github.com/codebydch/open-trickler-peripheral.git
sudo apt install memcached
pip3 install pymemcache

# Adafruit Blinka install
cd /code
pip3 install --upgrade adafruit-python-shell
wget https://raw.githubusercontent.com/adafruit/Raspberry-Pi-Installer-Scripts/master/raspi-blinka.py
sudo -E env PATH=$PATH python3 raspi-blinka.py
# say no to reboot, close everything down, and get ready for a reboot
rm raspi-blinka.py
# reboot
sudo reboot

# Adafruit Display install
cd /code
source venv/bin/activate
pip3 install adafruit-circuitpython-rgb-display
sudo apt-get install fonts-dejavu
sudo apt-get install python3-pil
sudo apt-get install python3-numpy

# NGINX Web Server Install and Setup
sudo apt install nginx
cd /code/open-trickler-peripheral/html
sudo cp *.html /var/www/html/
cd /code/open-trickler-peripheral/nginx/
sudo cp default /etc/nginx/sites-available/
# A stock Raspberry Pi OS image already links this; a tidied one may not.
sudo ln -sfn /etc/nginx/sites-available/default /etc/nginx/sites-enabled/default
sudo nginx -t && sudo systemctl reload nginx

# websocketd install
# linux_arm is 32-bit; use linux_arm64 on a 64-bit Raspberry Pi OS. Check with `uname -m`:
#   armv6l / armv7l -> websocketd-0.4.1-linux_arm.zip
#   aarch64         -> websocketd-0.4.1-linux_arm64.zip
cd /code
wget https://github.com/joewalnes/websocketd/releases/download/v0.4.1/websocketd-0.4.1-linux_arm.zip
unzip -d websocketd websocketd-0.4.1-linux_arm.zip
cd websocketd
sudo cp websocketd /usr/sbin

# No GPIO daemon is needed. The trickler motors and the servo both go through gpiozero,
# which picks its own pin factory (it prefers lgpio). pigpio, which this used to need,
# has been archived by its author and is not available from Trixie onwards.

# Charge history directory (the trickler creates it, but do it here so ownership is sane)
sudo mkdir -p /var/lib/opentrickler

# systemctl scripts install
cd /code/open-trickler-peripheral/system/
sudo cp *.service /etc/systemd/system/
sudo systemctl enable opentrickler_flask_app.service
sudo systemctl enable opentrickler_flask_servo_app.service
sudo systemctl enable opentrickler_screen.service
sudo systemctl enable opentrickler.service
sudo systemctl enable websocketd-1.service
sudo systemctl enable websocketd-2.service
sudo systemctl enable websocketd-4.service
sudo systemctl enable websocketd-5.service
sudo systemctl daemon-reload

# All Python dependencies in one go (the scripts do only this)
cd /code
source venv/bin/activate
pip install -r /code/open-trickler-peripheral/requirements-to-freeze.txt

# reboot
sudo reboot
